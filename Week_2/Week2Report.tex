function f = ueintbit(xa, ua, xb, ub)
    ubar = (ua+ub)/2;
    du = ub-ua;
    dx = xb-xa;
    f = (ubar^5 + (5/6)*ubar^3 * du^2 + (1/16) * ubar * du^4)*dx;
end
\end{lstlisting}

    for k=1:length(dudxs)
        re_l = re_ls(n);
        dudx = dudxs(k);
        laminar = true;
        integ = 0;
        int = 0;
        ils = 0;
        i=1;
        u_ = linspace(1, 1+dudx, np);

            i=i+1;
            integ = integ + ueintbit(x(i-1), u_(i-1), x(i), u_(i));
            theta_2(i) = (0.45/re_l) * u_(i)^(-6) * integ;
            m = -re_l * theta_2(i)  * dudx;
            H = thwaites_lookup(m);
            He = laminar_He(H);
            re_thet = re_l * u_(i) * sqrt(theta_2(i));

                laminar = false;    %transition
                int = i;       
        end
        if int ~= 0

            disp(['Natural transition at x = ' num2str(x(int)) ', Rethet/1000 = ' num2str(re_l * u_(int) * sqrt(theta_2(int)) / 1000)])
        else
            disp(['re_l = ' num2str(re_l) ', dudx = ' num2str(dudx)])

        end
        disp(' ')
    end
end

\begin{lstlisting}[language=Matlab, caption={The function thickdash which evaluates $d\mathbf{y}/dx$ for each time interval. (Exercise 4)}]
function dthickdx = thickdash(xmx0, thick)
    global re_l dudx u0
    dthickdx = zeros(2,1);
    He = thick(2)/thick(1);
    if He>=1.46
        H =(11*He + 15)/(48*He-59);
    else
        H = 2.803;        
    end 
    ue = u0 + dudx*xmx0;
    re_thet = re_l * ue * thick(1);
    cf = 0.091416*((H-1)*re_thet)^(-0.232) * exp(-1.26*H);
    cdiss = 0.010026*((H-1)*re_thet)^(-1/6);
    dthickdx(1,1) = cf/2 - ((H+2)/ue)*dudx*thick(1);
    dthickdx(2,1) = cdiss - (3/ue)*dudx*thick(2);
end
\end{lstlisting}

clear;
close all;

global re_l dudx u0

np = 101;
%for turbulent separation at x/L=1, dudx = -0.383
x = linspace(0,1,np); %length varies from 0->1
dudx = -0.383; %dimensionless gradient
re_l = 1e5;

theta = zeros(1,np);  %theta squared at each position
Hes = zeros(1,np);
Hes(1) = 1.57258;


laminar = true;
integ = 0;
int = 0; %location of laminar->turbulent transition
ils = 0; %location of laminar separation
itr = 0; %location of turbulent reattachment 
its = 0; %location of turbulent separation
i=1;
u_ = linspace(1, 1+dudx, np);
while laminar && i<np
    i=i+1;
    integ = integ + ueintbit(x(i-1), u_(i-1), x(i), u_(i));
    theta(i) = sqrt((0.45/re_l) * u_(i)^(-6) * integ);
    m = -re_l * theta(i)^2  * dudx;
    H = thwaites_lookup(m);
    Hes(i) = laminar_He(H);
    re_thet = re_l * u_(i) * theta(i);
    if log(re_thet) >= 18.4*Hes(i) - 21.74
        laminar = false;    %transition
        int = i;

    elseif m>= 0.09
        laminar = false;
        ils = i;
        Hes(i) = 1.51509;
    end            
end
delta_e = Hes(i)*theta(i);

while its == 0 && i<np %loop for turbulent boundary layer
    i=i+1;
    thick0 = [theta(i-1), delta_e];
    u0 = u_(i-1); %velocity at start of panle
    [delx, thickhist] = ode45(@thickdash, [0, x(i)-x(i-1)], thick0);
    theta(i) = thickhist(end, 1);
    delta_e = thickhist(end, 2);
    Hes(i) = delta_e/theta(i);
    if Hes(i)>1.58 && ils ~= 0
        itr = i; %turbulent reattachment
    end
    if Hes(i) <1.46
        its = i; %turbulent separation
    end
end

if Hes(i)>=1.46
        H =(11*Hes(i) + 15)/(48*Hes(i)-59);
else
    H = 2.803;        
end 
if its ~= 0
    Hes(its+1:np) = Hes(i);
    theta(its+1:np) = theta(i).*(u_(i)./u_(i+1:np)).^(H+2);
end


disp(['re_l = ' num2str(re_l) ', dudx = ' num2str(dudx)])
if int ~= 0
    disp(['Natural transition at x = ' num2str(x(int)) ', Rethet/1000 = ' num2str(re_l * u_(int) * theta(int) / 1000)])
end

    disp(['Separation at x = ' num2str(x(ils)) ', Rethet/1000 = ' num2str(re_l * u_(ils) * theta(ils) / 1000)])
end

end

end
