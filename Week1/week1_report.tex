function psixy = psipv(xc, yc, Gamma, x, y)
    %x and y can be matricies 
    %find the elementwise squared distance
    r_2 = (x-xc).^2 + (y-yc).^2; 
    %evaluate the streamfunction elementwise
    psixy = - (Gamma / (4 * pi)) * log(r_2);
end

function [infa, infb] = refpainf(del, X, Yin)
    if abs(Yin) <1e-6
        Y = 1e-6;
    else 
        Y = Yin;
    end 

    I0 = (-1/(4*pi)) * (X*log(X^2+Y^2) - (X-del)*log((X-del)^2 + Y^2) -2*del + 2*Y*(atan(X/Y)-atan((X-del)/Y)));
    I1 = (1/(8*pi))*((X^2+Y^2)*log(X^2+Y^2) - ((X-del)^2+Y^2)*log((X-del)^2+Y^2)-2*X*del+del^2);
    infa = (1-(X/del))*I0 - I1/del;
    infb = (X/del)*I0 + I1/del;
end

clear;
close all;
xmin = -2.5;
xmax = 2.5;
ymin = -2;
ymax = 2;
ny = 41;
nx = 51;
del = 1.5;
gamma_a = 0.5;

nv=100;
yc=0;
psi_approx = zeros(nx, ny);
infa = zeros(nx,ny) ;
infb = zeros(nx, ny);
xm = zeros(nx, ny);

psi_analy = zeros(nx,ny);


for i = 1:nx

        xm(i,j) = xmin + (i-1)*(xmax-xmin)/(nx-1);
        ym(i,j) = ymin + (j-1)*(ymax-ymin)/(ny-1);
        [infa(i,j), infb(i,j)] = refpainf(del, xm(i,j), ym(i,j));
        psi_analy(i,j) = gamma_a*infa(i,j) + gamma_b *infb(i,j);
            
    end
end
for n = 1:nv

    Gamma = (gamma_a*(del-xc)/del + gamma_b*xc/del)*del/nv;
    for i = 1:nx
        for j = 1:ny
            psi_approx(i,j) = psi_approx(i,j) + psipv(xc, yc, Gamma, xm(i,j), ym(i,j));
        end 
    end
end

function [infa, infb] = panelinf(xa, ya, xb, yb, x, y)
    del = norm([(xb-xa), (yb-ya)]);
    r = [(x-xa), (y-ya)];
    t = [(xb-xa), (yb-ya)]/del;
    n = [(yb-ya), -(xb-xa)]/del;
    X = dot(r, t);
    Y = dot(r, n);
    [infa, infb] = refpainf(del, X, Y);
end

close all;

xmax = 5;
ymin = 0;
ymax = 4;

for i = 1:nx

        xm(i,j) = xmin + (i-1)*(xmax-xmin)/(nx-1);
        ym(i,j) = ymin + (j-1)*(ymax-ymin)/(ny-1);
        [infa(i,j), infb(i,j)] = panelinf(xa, ya, xb, yb, xm(i,j), ym(i,j));
        psi_analy(i,j) = gamma_a*infa(i,j) + gamma_b *infb(i,j);
            
    end
end


l=0;
panel_length = norm([(xb-xa), (yb-ya)]);
dl = panel_length/nv; 
pl_vec = [(xb-xa), (yb-ya)]/norm([(xb-xa), (yb-ya)]);
a = [xa,ya];
b = [xb, yb];
for n = 1:nv
    centre = a + ((2*n-1)/2)*dl*pl_vec;
    Gamma = (gamma_a*norm(b-centre)/panel_length + gamma_b*norm(centre-a)/panel_length)*dl;
    for i = 1:nx

            psi_approx(i,j) = psi_approx(i,j) + psipv(centre(1), centre(2), Gamma, xm(i,j), ym(i,j));
        end 
    end
end

\begin{lstlisting}[language=Matlab, caption=Code for solving the panel method around a cylinder with an angle of attack (Exercise 5)]
function lhsmat = build_lhs(xs, ys)
    np = length(xs)-1;
    psip = zeros(np, np+1);
    infa = zeros(np, np+1);
    infb = zeros(np, np+1);

    for i=1:np %special case for j=1 
        [infa(i,1), infb(i,1)] = panelinf(xs(1), ys(1), xs(2), ys(2), xs(i), ys(i));
        psip(i, 1) = infa(i,1);
    end
    for j=2:np % main loop to generate psip matrix
        for i=1:np
            [infa(i, j), infb(i,j)] = panelinf(xs(j), ys(j), xs(j+1), ys(j+1), xs(i), ys(i));
            psip(i,j) = infa(i,j) + infb(i,j-1);
        end
    end

    for i =1:np %special case for j = np+1
        psip(i,np+1) = infb(i,j-1);
    end
    lhsmat = zeros(np+1, np+1);
    for i=1:np-1
        for j=1:np+1
            lhsmat(i,j) = psip(i+1, j) - psip(i,j);
        end
    end
    %set up the equation that gamma_np+1 = gamma_1
    lhsmat(np, 1) = 1;
    lhsmat(np, np+1) = -1;

    %set up the equation that gamma_1 = 0
    lhsmat(np+1,1)=1;

end

function rhsvec = build_rhs(xs,ys, alpha)
    np = length(xs)-1; 
    rhsvec = zeros(np+1, 1); %column vector b 
    psi_fs = zeros(np,1);

    for i = 1:np
        psi_fs(i,1) = ys(i)*cos(alpha) - xs(i)*sin(alpha); %free stream streamfunction
    end 
    for i = 1:np-1
        rhsvec(i,1) = psi_fs(i,1)-psi_fs(i+1,1);
    end
    rhsvec(np,1)=0;
    rhsvec(np+1,1)=0;
    %keep b(np) and b(np+1)=0 such that gamma_0=1 and gamma_0 = gamma_np+1
end

clear;
xmin = -2.5;
xmax = 2.5;
ymin = -2;
ymax = 2;
ny = 41;
nx = 51;

psi = zeros(nx, ny);
psi_fs = zeros(nx,ny);

    xs(i) = cos(theta(i));
    ys(i) = sin(theta(i));
end


A = build_lhs(xs, ys);
b = build_rhs(xs, ys, alpha);
gam = A\b;


for i = 1:nx
    for j = 1:ny
        xm(i,j) = xmin + (i-1)*(xmax-xmin)/(nx-1);
        ym(i,j) = ymin + (j-1)*(ymax-ymin)/(ny-1);
        psi(i,j) = ym(i,j)*cos(alpha) - xm(i,j)*sin(alpha); %free stream steamfunction
        for s = 1:np
            [infa, infb] = panelinf(xs(s), ys(s), xs(s+1), ys(s+1), xm(i,j), ym(i,j));
            psi(i,j) = psi(i,j) + gam(s)*infa + gam(s+1) *infb;
        end
    end
end


c = -1.75:0.25:1.75;
