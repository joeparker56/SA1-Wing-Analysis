function psixy = psipv(xc, yc, Gamma, x, y)
    %x and y can be matricies 
    %find the elementwise squared distance
    r_2 = (x-xc).^2 + (y-yc).^2; 
    %evaluate the streamfunction elementwise
    psixy = - (Gamma / (4 * pi)) * log(r_2);
end
\end{lstlisting}

\begin{lstlisting}[language=Matlab, caption={Function for evaluating influence coefficients $f_a$, $f_b$ at an arbitrary point due to a reference vortex sheet which starts at $(0,0)$, has length $\Delta$ and is along the x-axis. (Exercise 2)}]
function [infa, infb] = refpainf(del, X, Yin)
    if abs(Yin) <1e-6
        Y = 1e-6;
    else 
        Y = Yin;
    end 

    I0 = (-1/(4*pi)) * (X*log(X^2+Y^2) - (X-del)*log((X-del)^2 + Y^2) -2*del + 2*Y*(atan(X/Y)-atan((X-del)/Y)));
    I1 = (1/(8*pi))*((X^2+Y^2)*log(X^2+Y^2) - ((X-del)^2+Y^2)*log((X-del)^2+Y^2)-2*X*del+del^2);
    infa = (1-(X/del))*I0 - I1/del;
    infb = (X/del)*I0 + I1/del;
end
\end{lstlisting}

\begin{lstlisting}[language=Matlab, caption=Function for evaluating the influence coefficients for any vortex-sheet-panel at an arbitrary point in space. (Exercise 3)]
function [infa, infb] = panelinf(xa, ya, xb, yb, x, y)
    del = norm([(xb-xa), (yb-ya)]); %find the length of the sheet panel
    r = [(x-xa), (y-ya)]; %define the vector from the start of the sheet panel to the arbitrary point (x,y) 
    t = [(xb-xa), (yb-ya)]/del; %find the unit vector tangential to the sheet panel
    n = [(yb-ya), -(xb-xa)]/del; %find the unit vector normal to the sheet panel
    X = dot(r, t); %find the tangential distance from the sheet panel
    Y = dot(r, n); %find the perpendicular distance from the sheet panel
    [infa, infb] = refpainf(del, X, Y); %evaluate the streamfunction using the reference sheet panel function
end 
\end{lstlisting}

\begin{lstlisting}[language=Matlab, caption=Function for generating the $\mathbf{A}$ matrix to solve the panel method problem. (Exercise 5)]
function lhsmat = build_lhs(xs, ys)
    np = length(xs)-1;
    psip = zeros(np, np+1);
    infa = zeros(np, np+1);
    infb = zeros(np, np+1);

    for i=1:np %special case for j=1 
        [infa(i,1), infb(i,1)] = panelinf(xs(1), ys(1), xs(2), ys(2), xs(i), ys(i));
        psip(i, 1) = infa(i,1);
    end
    for j=2:np % main loop to generate psip matrix
        for i=1:np
            [infa(i, j), infb(i,j)] = panelinf(xs(j), ys(j), xs(j+1), ys(j+1), xs(i), ys(i));
            psip(i,j) = infa(i,j) + infb(i,j-1);
        end
    end

    for i =1:np %special case for j = np+1
        psip(i,np+1) = infb(i,j-1);
    end
    lhsmat = zeros(np+1, np+1);
    for i=1:np-1
        for j=1:np+1
            lhsmat(i,j) = psip(i+1, j) - psip(i,j);
        end
    end
    %set up the equation that gamma_np+1 = gamma_1
    lhsmat(np, 1) = 1;
    lhsmat(np, np+1) = -1;

    %set up the equation that gamma_1 = 0
    lhsmat(np+1,1)=1;

end
\end{lstlisting}
\begin{lstlisting}[language=Matlab, caption=Function for generating the $\mathbf{b}$ vector for the panel method problem. (Exercise 5)]
function rhsvec = build_rhs(xs,ys, alpha)
    np = length(xs)-1; 
    rhsvec = zeros(np+1, 1); %column vector b 
    psi_fs = zeros(np,1);

    for i = 1:np
        psi_fs(i,1) = ys(i)*cos(alpha) - xs(i)*sin(alpha); %free stream streamfunction
    end 
    for i = 1:np-1
        rhsvec(i,1) = psi_fs(i,1)-psi_fs(i+1,1);
    end
    rhsvec(np,1)=0;
    rhsvec(np+1,1)=0;
    %keep b(np) and b(np+1)=0 such that gamma_0=1 and gamma_0 = gamma_np+1
end
\end{lstlisting}
\begin{lstlisting}[language=Matlab, caption=Script for solving the panel method problem for a cylinder at an angle of attack. (Exercise 5)]
clear;
xmin = -2.5;
xmax = 2.5;
ymin = -2;
ymax = 2;
ny = 41;
nx = 51;

psi = zeros(nx, ny);
psi_fs = zeros(nx,ny);
xm = zeros(nx, ny);
ym = zeros(nx, ny);
alpha = pi/24;

np = 100;
theta = (0:np)*2*pi/np;
xs = zeros(np+1);
ys = zeros(np+1);

for i = 1:np+1
    %points around the cylinder
    xs(i) = cos(theta(i));
    ys(i) = sin(theta(i));
end

%solve the panel method problem by generating matricies and using gam = A^-1 b
A = build_lhs(xs, ys);
b = build_rhs(xs, ys, alpha);
gam = A\b;


for i = 1:nx
    for j = 1:ny
        xm(i,j) = xmin + (i-1)*(xmax-xmin)/(nx-1);
        ym(i,j) = ymin + (j-1)*(ymax-ymin)/(ny-1);
        psi(i,j) = ym(i,j)*cos(alpha) - xm(i,j)*sin(alpha); %free stream steamfunction
        for s = 1:np
            %evaluate the influence coefficients and streamfunction
            [infa, infb] = panelinf(xs(s), ys(s), xs(s+1), ys(s+1), xm(i,j), ym(i,j));
            psi(i,j) = psi(i,j) + gam(s)*infa + gam(s+1) *infb;
        end
    end
end


c = -1.75:0.25:1.75;
circ =0;
for i=1:np
    %calculate the total circulation around the cylinder
    circ = circ + gam(i)*2*pi/np;
end

contour(xm,ym,psi,c);
xlabel('x');
ylabel('y');
title('Contour plot of flow around cylinder for alpha = pi/24');

hold on
plot(xs,ys)
hold off
